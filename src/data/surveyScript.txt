/**
 * МалышДок — опрос для родителей.
 * Создаёт Google Форму со всеми вопросами и переходами между разделами.
 * Как запустить: script.google.com → Новый проект → вставить этот код → Сохранить →
 * выбрать функцию createSurvey → Выполнить → разрешить доступ. Ссылки появятся в «Журнале выполнения».
 */
function createSurvey() {
  var form = FormApp.create("МалышДок: опрос для родителей");
  form.setDescription("Здравствуйте! Мы делаем МалышДок — бесплатный помощник для родителей: первая помощь, когда срочно к врачу, дозы лекарств по весу, календарь прививок и осмотров, дневник болезни.\n\nПомогите нам сделать его полезнее — опрос займёт 5–7 минут. Ответы анонимны, обязательные вопросы отмечены звёздочкой.");
  form.setConfirmationMessage("Спасибо! Ваши ответы помогут сделать МалышДок удобнее для тысяч родителей.");
  form.setProgressBar(true);
  form.setCollectEmail(false);
  form.setAllowResponseEdits(false);
  form.setLimitOneResponsePerUser(false);

  function opt(item, q) {
    item.setTitle(q.q);
    if (q.help) item.setHelpText(q.help);
    if (q.req) item.setRequired(true);
    return item;
  }
  function add(q) {
    var it;
    switch (q.t) {
      case 'radio':
        it = opt(form.addMultipleChoiceItem(), q).setChoiceValues(q.opts);
        if (q.other) it.showOtherOption(true);
        return it;
      case 'checkbox':
        it = opt(form.addCheckboxItem(), q).setChoiceValues(q.opts);
        if (q.other) it.showOtherOption(true);
        if (q.max) it.setValidation(FormApp.createCheckboxValidation().setHelpText('Выберите не больше ' + q.max).requireSelectAtMost(q.max).build());
        return it;
      case 'scale':
        return opt(form.addScaleItem(), q).setBounds(q.lo, q.hi).setLabels(q.lo_l, q.hi_l);
      case 'grid':
        return opt(form.addGridItem(), q).setRows(q.rows).setColumns(q.cols);
      case 'paragraph':
        return opt(form.addParagraphTextItem(), q);
      case 'email':
        return opt(form.addTextItem(), q).setValidation(FormApp.createTextValidation().setHelpText('Похоже, в адресе ошибка').requireTextIsEmail().build());
      case 'branch':
        return opt(form.addMultipleChoiceItem(), q);
    }
  }

  var pages = {};
  var branches = [];
  // ===== О вас и вашем ребёнке =====
  add({"t": "radio", "q": "Кем вы приходитесь ребёнку?", "req": true, "opts": ["Мама", "Папа", "Бабушка или дедушка"], "other": true});
  add({"t": "radio", "q": "Сколько у вас детей до 18 лет?", "req": true, "opts": ["1", "2", "3 и больше"]});
  add({"t": "checkbox", "q": "Сколько лет вашим детям?", "help": "Отметьте все подходящие варианты", "req": true, "opts": ["До 1 года", "1–3 года", "3–7 лет", "7–12 лет", "12–18 лет"]});
  add({"t": "radio", "q": "Где вы живёте?", "req": true, "opts": ["Город-миллионник", "Крупный город (от 250 тыс.)", "Небольшой город", "Посёлок или село"]});
  add({"t": "checkbox", "q": "Ребёнок внезапно заболел. Куда вы обращаетесь в первую очередь?", "help": "Можно выбрать несколько", "req": true, "opts": ["Пишу или звоню знакомому педиатру", "Вызываю врача из поликлиники", "Звоню в скорую (103)", "Ищу в интернете", "Спрашиваю в чатах и сообществах родителей", "Советуюсь с родственниками", "Смотрю в приложениях", "Смотрю в книгах"], "other": true});
  add({"t": "scale", "q": "Насколько уверенно вы действуете, когда ребёнок внезапно заболел?", "req": true, "lo": 1, "hi": 5, "lo_l": "Теряюсь", "hi_l": "Знаю, что делать"});
  add({"t": "checkbox", "q": "Что даётся вам труднее всего?", "help": "Выберите до 3 вариантов", "req": true, "max": 3, "opts": ["Понять, когда пора срочно к врачу", "Рассчитать дозу лекарства", "Сбить температуру", "Разобраться, что за сыпь", "Не пропустить прививки", "Помнить про плановые осмотры", "Понять, нормально ли ребёнок развивается", "Следить за аптечкой и сроками годности", "Хранить справки и документы", "Вспомнить на приёме, чем и как болел ребёнок"], "other": true});
  branches.push({ item: add({"t": "branch", "q": "Вы уже пользовались приложением МалышДок?", "req": true}), opts: [["Да, пользуюсь регулярно", "users"], ["Да, открывал(а) пару раз", "users"], ["Нет, слышу впервые", "new"]] });

  // ===== Ваш опыт с МалышДок =====
  pages["users"] = form.addPageBreakItem().setTitle("Ваш опыт с МалышДок").setHelpText("Несколько вопросов о том, как вам приложение.");
  add({"t": "radio", "q": "Как часто вы открываете МалышДок?", "req": true, "opts": ["Почти каждый день", "Раз в неделю", "Раз в месяц", "Только когда ребёнок болеет", "Реже"]});
  add({"t": "checkbox", "q": "Какими разделами вы пользовались?", "help": "Отметьте все", "req": true, "opts": ["Первая помощь", "Красные флаги", "Сыпь", "Дневник болезни", "Аптечка и дозы", "Развитие", "Вакцинация", "Осмотры", "Врачи", "Документы", "Совет дня", "Отчёт для врача"]});
  add({"t": "radio", "q": "Какой раздел для вас самый полезный?", "req": true, "opts": ["Первая помощь", "Красные флаги", "Сыпь", "Дневник болезни", "Аптечка и дозы", "Развитие", "Вакцинация", "Осмотры", "Врачи", "Документы", "Совет дня", "Отчёт для врача"]});
  add({"t": "scale", "q": "Насколько удобно пользоваться приложением?", "req": true, "lo": 1, "hi": 5, "lo_l": "Очень неудобно", "hi_l": "Очень удобно"});
  add({"t": "scale", "q": "Насколько вы доверяете информации в приложении?", "req": true, "lo": 1, "hi": 5, "lo_l": "Не доверяю", "hi_l": "Полностью доверяю"});
  add({"t": "paragraph", "q": "Расскажите о случае, когда МалышДок вам помог", "help": "Если такого не было, пропустите вопрос"});
  add({"t": "paragraph", "q": "Что было непонятно, неудобно или раздражало?"});
  add({"t": "checkbox", "q": "Чего вам не хватает в приложении?", "help": "Можно выбрать несколько", "opts": ["Напоминаний о приёме лекарств", "Общего доступа для второго родителя", "Онлайн-консультации с педиатром", "Записи к врачу", "Дневника сна и питания", "Графиков роста и веса", "Экспорта данных в PDF", "Советов для подростков"], "other": true});
  add({"t": "scale", "q": "Насколько вероятно, что вы порекомендуете МалышДок знакомым родителям?", "req": true, "lo": 0, "hi": 10, "lo_l": "Точно нет", "hi_l": "Обязательно"});

  // ===== Ваши привычки и ожидания =====
  pages["new"] = form.addPageBreakItem().setTitle("Ваши привычки и ожидания").setHelpText("Расскажите, как вы справляетесь сейчас и что было бы полезно.");
  add({"t": "radio", "q": "Пользуетесь ли вы сейчас приложениями о здоровье ребёнка?", "req": true, "opts": ["Да, пользуюсь", "Пробовал(а), но бросил(а)", "Нет"]});
  add({"t": "checkbox", "q": "Где вы храните информацию о прививках и осмотрах ребёнка?", "req": true, "opts": ["Бумажная карта или прививочный сертификат", "Госуслуги", "Заметки или фото в телефоне", "Специальное приложение", "Держу в голове", "Нигде, всё в поликлинике"], "other": true});
  add({"t": "grid", "q": "Насколько вам была бы полезна каждая функция?", "req": true, "rows": ["Пошаговая первая помощь", "Красные флаги: когда срочно к врачу", "Калькулятор дозы лекарства по весу", "Календарь прививок с напоминаниями", "Подсказки, что за сыпь", "Дневник болезни: температура и лекарства", "Аптечка со сроками годности", "Нормы развития по возрасту", "Отчёт для врача одним файлом", "Работа без интернета"], "cols": ["Не нужно", "Возможно", "Очень нужно"]});
  add({"t": "checkbox", "q": "Что для вас важнее всего в таком приложении?", "help": "Выберите до 3 вариантов", "req": true, "max": 3, "opts": ["Информация проверена врачами", "Просто и быстро найти нужное", "Работает без интернета", "Бесплатно", "Мои данные никуда не передаются", "Можно вести нескольких детей", "Нет рекламы"]});
  add({"t": "checkbox", "q": "Что может остановить вас от использования?", "help": "Можно выбрать несколько", "opts": ["Не доверяю медицинским советам из приложений", "Не хочу вводить данные о ребёнке", "Нет места на телефоне", "Не хочу ещё одно приложение", "Всё равно пойду к врачу"], "other": true});
  add({"t": "radio", "q": "Хотели бы вы попробовать МалышДок?", "req": true, "opts": ["Да, прямо сейчас", "Да, если будет бесплатно", "Возможно позже", "Нет"]});

  // ===== Последние вопросы =====
  pages["final"] = form.addPageBreakItem().setTitle("Последние вопросы");
  add({"t": "radio", "q": "Были бы вы готовы платить за расширенные функции (например, онлайн-консультации или общий доступ для семьи)?", "req": true, "opts": ["Нет, пользуюсь только бесплатным", "До 99 ₽ в месяц", "100–299 ₽ в месяц", "300 ₽ в месяц и больше", "Предпочту разовую покупку"]});
  add({"t": "paragraph", "q": "Что ещё вы хотели бы нам сказать?"});
  add({"t": "email", "q": "Оставьте email, если хотите первыми тестировать новые функции", "help": "Необязательно. Указывая email, вы соглашаетесь получать письма от команды МалышДок. Отписаться можно в любой момент."});

  branches.forEach(function (b) {
    b.item.setChoices(b.opts.map(function (o) { return b.item.createChoice(o[0], pages[o[1]]); }));
  });
  pages["new"].setGoToPage(pages["final"]);

  Logger.log('Форма готова!');
  Logger.log('Ссылка для родителей: ' + form.getPublishedUrl());
  Logger.log('Редактирование и ответы: ' + form.getEditUrl());
}
